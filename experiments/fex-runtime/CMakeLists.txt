cmake_minimum_required(VERSION 3.22)
project(ARMXBOXFexProbe LANGUAGES C CXX ASM)
# Intentionally separate from the production Android C++17 build.
option(ARMXBOX_HOST_SIMULATOR "Run generated ARM64 under VIXL on x86-64 Linux (not device proof)" OFF)
if(ARMXBOX_HOST_SIMULATOR AND (ANDROID OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$"))
  message(FATAL_ERROR "Host simulation requires x86-64 Linux; use the Android target for native ARM64")
endif()
if(NOT ARMXBOX_HOST_SIMULATOR AND (NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a"))
  message(FATAL_ERROR "This probe requires the Android ARM64 NDK toolchain")
endif()
set(FEX_SOURCE_DIR "" CACHE PATH "Checkout of the pinned FEX source")
set(FEX_PIN 4e2b509a42b7762a48c407d6a59bb897dca1ded8)
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${FEX_SOURCE_DIR}" rev-parse HEAD
  OUTPUT_VARIABLE FEX_ACTUAL OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT FEX_ACTUAL STREQUAL FEX_PIN)
  message(FATAL_ERROR "FEX checkout must be ${FEX_PIN}, got ${FEX_ACTUAL}")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BUILD_TESTING OFF)
set(BUILD_STEAM_SUPPORT ON) # Suppress upstream install rules, not a Steam dependency.
set(ENABLE_FEX_ALLOCATOR ON)
set(ENABLE_JEMALLOC_GLIBC_ALLOC OFF) # Bionic has no glibc; Linux thunks are excluded.
set(ENABLE_OFFLINE_TELEMETRY OFF)
if(ARMXBOX_HOST_SIMULATOR)
  set(ARCHITECTURE_x86_64 1)
  set(ENABLE_VIXL_SIMULATOR ON)
  add_compile_definitions(ARCHITECTURE_x86_64=1 VIXL_SIMULATOR=1 VIXL_INCLUDE_SIMULATOR_AARCH64=1)
  add_compile_options(-mcx16 -mavx)
else()
  set(ARCHITECTURE_arm64 1)
  add_compile_definitions(ARCHITECTURE_arm64=1)
endif()
add_compile_definitions(_LARGEFILE64_SOURCE)
include(CheckCXXSourceCompiles)
include(CheckCXXCompilerFlag)
set(CMAKE_REQUIRED_FLAGS "-Werror=attributes")
check_cxx_source_compiles("__attribute__((preserve_all)) int f(){return 1;} int main(){return f();}" HAS_CLANG_PRESERVE_ALL)
unset(CMAKE_REQUIRED_FLAGS)
if(HAS_CLANG_PRESERVE_ALL)
  add_compile_definitions("FEX_PRESERVE_ALL_ATTR=__attribute__((preserve_all))" FEX_HAS_PRESERVE_ALL_ATTR=1)
else()
  add_compile_definitions(FEX_PRESERVE_ALL_ATTR= FEX_HAS_PRESERVE_ALL_ATTR=0)
endif()
if(NOT ARMXBOX_HOST_SIMULATOR)
  set(FEX_TUNE_COMPILE_FLAGS -march=armv8-a+crc -ffixed-x18)
endif()
set(GIT_HASH "${FEX_PIN}")
set(GIT_DESCRIBE_STRING "ARMXBOX-p0-4e2b509a")
string(REGEX REPLACE "(..)" "0x\\1, " GIT_HASH_ARRAY "${GIT_HASH}")
list(APPEND CMAKE_MODULE_PATH "${FEX_SOURCE_DIR}/Data/CMake")
include(LinkerGC)
set(BUILD_SHARED_LIBS OFF)
set(FMT_INSTALL OFF)
set(XXHASH_BUNDLED_MODE ON)
set(XXHASH_BUILD_XXHSUM OFF)
foreach(dep unordered_dense fmt rpmalloc SoftFloat-3e cephes)
  add_subdirectory("${FEX_SOURCE_DIR}/External/${dep}" "external/${dep}")
endforeach()
add_subdirectory("${FEX_SOURCE_DIR}/External/xxhash/cmake_unofficial" external/xxhash)
add_subdirectory("${FEX_SOURCE_DIR}/FEXHeaderUtils" FEXHeaderUtils)
add_subdirectory("${FEX_SOURCE_DIR}/CodeEmitter" CodeEmitter)
if(ARMXBOX_HOST_SIMULATOR)
  add_subdirectory("${FEX_SOURCE_DIR}/External/vixl" external/vixl)
endif()
add_subdirectory("${FEX_SOURCE_DIR}/FEXCore" FEXCore)
# Force the entire core to link, exposing unresolved Bionic dependencies.
add_library(armxbox_fex_link_probe SHARED link_probe.cpp)
target_link_libraries(armxbox_fex_link_probe PRIVATE
  -Wl,--whole-archive FEXCore FEXCore_Base -Wl,--no-whole-archive JemallocLibs)
target_link_options(armxbox_fex_link_probe PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)

add_library(armxbox_fex_diagnostic SHARED diagnostic.cpp)
target_link_libraries(armxbox_fex_diagnostic PRIVATE FEXCore FEXCore_Base JemallocLibs)
target_compile_options(armxbox_fex_diagnostic PRIVATE -fno-exceptions)
target_link_options(armxbox_fex_diagnostic PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
if(ANDROID)
  target_sources(armxbox_fex_diagnostic PRIVATE jni.cpp abi_witness.S)
  target_link_libraries(armxbox_fex_diagnostic PRIVATE log)
else()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/abi_witness.bin"
    COMMAND "${CMAKE_C_COMPILER}" --target=aarch64-linux-gnu -c
      "${CMAKE_CURRENT_SOURCE_DIR}/abi_witness.S" -o "${CMAKE_CURRENT_BINARY_DIR}/abi_witness.o"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/extract-text.py"
      "${CMAKE_CURRENT_BINARY_DIR}/abi_witness.o" "${CMAKE_CURRENT_BINARY_DIR}/abi_witness.bin"
    DEPENDS abi_witness.S scripts/extract-text.py VERBATIM)
  add_custom_target(armxbox_abi_blob DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/abi_witness.bin")
  add_executable(armxbox_abi_witness_test abi_witness_test.cpp)
  target_link_libraries(armxbox_abi_witness_test PRIVATE vixl::vixl)
  add_dependencies(armxbox_abi_witness_test armxbox_abi_blob)
  add_executable(armxbox_division_native_test division_native_test.cpp)
  add_executable(armxbox_fex_selftest main.cpp)
  target_link_libraries(armxbox_fex_selftest PRIVATE armxbox_fex_diagnostic)
endif()
